% Cuestión X. Mejoras y extensiones futuras de stegobmp (ninguna implementada; casi todas rompen el formato de la cátedra).

\begin{enumerate}
\item \textbf{Cifrado autenticado} (AES-GCM~\cite{nist80038d} o un HMAC): hoy una contraseña equivocada o un archivo alterado pueden pasar inadvertidos.
\item \textbf{Sal aleatoria y más iteraciones de PBKDF2}~\cite{nist800132}: la sal fija hace que la misma contraseña dé siempre la misma clave e IV.
\item \textbf{Posiciones de inserción derivadas de la clave}~\cite{sharp2001}, en lugar de recorrer el portador en orden desde el primer byte.
\item \textbf{LSB matching} ($\pm1$) en lugar de reemplazo, para resistir el ataque de $\chi^2$~\cite{westfeld2000}.
\item \textbf{Modo de análisis} (\texttt{-analyze}, \texttt{-capacity}) que integre las herramientas usadas en el estegoanálisis.
\item \textbf{Más portadores}: BMP V4/V5, 32 bits y PNG.
\item \textbf{Leer la contraseña de la terminal} sin que quede en el historial ni en la lista de procesos.
\end{enumerate}
